# Project 1: Text RPG Engine with Conditional Branching
class Hero:
    def __init__(self, name, role):
        self.name = name
        self.hp = 100
        self.inventory = []

    def equip(self, item):
        self.inventory.append(item)
        print(f"[+] Equipped {item}!")

player = Hero("Amina", "Cyber-Mage")
player.equip("Photon Blade")
if "Photon Blade" in player.inventory:
    print(">> Vault door breached. Mission complete.")
# Project 2: Pygame 2D Arcade Engine Loop
import pygame, sys

pygame.init()
screen = pygame.display.set_mode((640, 480))
clock = pygame.time.Clock()
ship_pos = [320, 400]
velocity = 5

while True:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            sys.exit()
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]: ship_pos[0] -= velocity
    if keys[pygame.K_RIGHT]: ship_pos[0] += velocity
    screen.fill((15, 23, 42))
    pygame.draw.rect(screen, (244, 96, 12), (*ship_pos, 30, 15))
    pygame.display.flip()
    clock.tick(60)
# Project 3: Cryptographic Cipher Automation
def encrypt_message(text, shift):
    result = ""
    for char in text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            result += chr((ord(char) - base + shift) % 26 + base)
        else:
            result += char
    return result

secret = encrypt_message("STEMulus Innovator 2026", 4)
print(f"Encrypted payload: {secret}")

Structured Curriculum Architecture

10-Week Engineering Syllabus

Each week includes a 1-on-1 private lesson with your dedicated mentor, weekly code reviews, and a tangible software deliverable for your student portfolio.

PHASE 01

Syntax, Variables & Algorithmic Logic

Weeks 1 to 3 · Transitioning from visual blocks to precision syntax

Week 01
Development Setup & I/O

Installing VS Code and Python. Mastering print statements, dynamic user input, and string concatenation.

Week 02
Variables & Arithmetic Operators

Integers, floats, booleans, and math expressions. Building an interactive currency and metric converter.

Week 03
Conditionals & Decision Trees

If, elif, else logic branches, comparison operators, and building a multi-outcome interactive quest game.

PHASE 02

Data Collections, Iteration & Functions

Weeks 4 to 5 · Managing complex information and modular design

Week 04
Lists, Dictionaries & Loops

For and while loops, list manipulation, dictionary key-value lookups, and inventory management systems.

Week 05
Modular Functions & Scope

Defining reusable functions, parameter passing, return values, and building automated mathematical utilities.

PHASE 03

Pygame Arcade Engineering & Physics

Weeks 6 to 8 · Building graphic 2D arcade games from scratch

Week 06
Pygame Canvas & Game Loops

Initializing graphics windows, coordinate planes, clock tick rates, and event listening pipelines.

Week 07
Sprites, Animation & Movement

Keyboard control integration, vector speed, drawing shapes, and loading custom pixel assets.

Week 08
Collision Detection & Audio

Hitbox calculation, score keeping, sound effect triggers, and game-over state machines.

PHASE 04

Capstone Portfolio Project & Graduation

Weeks 9 to 10 · Independent project creation and technical showcase

Week 09
Capstone Architecture & Coding

Student designs their own original game or software tool under the direct 1-on-1 guidance of their mentor.

Week 10
Code Review, Polish & Presentation

Debugging edge cases, code refactoring, publishing to GitHub, and live demonstration for parents.
