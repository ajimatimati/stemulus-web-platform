// React + TypeScript: Dynamic Project Data Fetching
import React, { useState, useEffect } from 'react';

export const ProjectDashboard = () => {
  const [projects, setProjects] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('/api/v1/projects')
      .then(res => res.json())
      .then(data => {
        setProjects(data);
        setLoading(false);
      });
  }, []);

  return (
    <div className="project-feed">
      {loading ? <span>Synchronizing...</span> :
        projects.map(p => <Card key={p.id} title={p.name} />)
      }
    </div>
  );
};
// Node.js / Express REST API Controller
const express = require('express');
const { pool } = require('./db');
const router = express.Router();

router.get('/projects', async (req, res) => {
  try {
    const query = 'SELECT id, name, status, created_at FROM projects ORDER BY created_at DESC';
    const { rows } = await pool.query(query);
    res.status(200).json(rows);
  } catch (err) {
    res.status(500).json({ error: 'Database query fault' });
  }
});

module.exports = router;
-- PostgreSQL Relational Schema Architecture
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(64) UNIQUE NOT NULL,
    email VARCHAR(128) UNIQUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE projects (
    id SERIAL PRIMARY KEY,
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(128) NOT NULL,
    status VARCHAR(32) DEFAULT 'development',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

Structured Software Engineering Pathway

16-Week Full-Stack Syllabus

Students build, integrate, and deploy two production web applications with 1-on-1 mentor guidance, comprehensive code reviews, and industry toolchains.

PHASE 01

Modern JavaScript (ES6+) & React Fundamentals

Weeks 1 to 4 · Asynchronous operations, component hierarchy, props, and state

Weeks 01-02
Advanced JavaScript & DOM

Promises, async/await, closures, array methods (map, filter, reduce), and modern modular architecture.

Weeks 03-04
React Components & Hooks

JSX syntax, useState, useEffect, controlled inputs, and decomposing UIs into reusable component hierarchies.

PHASE 02

Backend Architecture, Node.js & REST APIs

Weeks 5 to 8 · Server runtime, routing protocols, middleware, and request lifecycles

Weeks 05-06
Node.js Runtime & Express

Event-driven architecture, npm package management, building HTTP endpoints, and request/response pipelines.

Weeks 07-08
RESTful Architecture & CRUD

Standard HTTP methods (GET, POST, PUT, DELETE), status codes, body validation, and error-handling middleware.

PHASE 03

Relational Databases & PostgreSQL Modeling

Weeks 9 to 12 · SQL syntax, schema design, table relationships, and data persistence

Weeks 09-10
Relational Schemas & SQL

Designing database tables, primary/foreign keys, one-to-many relationships, and complex JOIN queries.

Weeks 11-12
Database Integration & Security

Connecting Node.js to PostgreSQL via connection pooling, parameterized queries to prevent SQL injection.

PHASE 04

Authentication, Cloud Deployment & Capstone

Weeks 13 to 16 · JWT auth, cloud hosting, CI/CD, and shipping the final web application

Weeks 13-14
User Authentication & Sessions

Password hashing with bcrypt, JSON Web Tokens (JWT), protected routes, and role-based access control.

Weeks 15-16
Cloud Deployment & Portfolio Defense

Deploying to Render/Vercel with managed PostgreSQL, custom domain configuration, and portfolio presentation.
